func TestIndependentSingleReceiptVotes(t *testing.T) {
 a,b:=testTx(1),testTx(2)
 reports:=[]*types.LocalOrder{localOrder(0,a,b),localOrder(1,b,a),localOrder(2,a),localOrder(3,b)}
 dm:=NewDependencyManager(); dm.BuildGraphAndClassifyTxs(reports,5,1,1,nil);dm.CutShadedTail()
 // The four reports vote AB, BA, AB, BA. Both vertices occur 3 times.
 if dm.weights[a][b]!=2 || dm.weights[b][a]!=2 || !graphHasEdge(dm.graph,a,b) || !IsTournament(dm.graph){t.Fatal("dissertation weight rule violated")}
 t.Log("paper: presence=(3,3), weights=(2,2), ID tie A<B -> A->B; current agrees; co-presence-only would give (1,1) and no edge")
}

func TestIndependentFractionalGammaLemmaGap(t *testing.T) {
 a,b:=testTx(1),testTx(2)
 if err:=ValidateThemisParameters(6,1,.9);err!=nil{t.Fatal(err)}
 reports:=[]*types.LocalOrder{localOrder(0,a,b),localOrder(1,a,b),localOrder(2,b,a),localOrder(3,b,a),localOrder(4)}
 dm:=NewDependencyManager();dm.BuildGraphAndClassifyTxs(reports,6,1,.9,nil);dm.CutShadedTail()
 // n(2gamma-1)=4.8>4; n-2f=4. Literal n(1-gamma)+f+1=2.6,
 // so the integer threshold is 3. Four occurrences each, votes 2:2.
 // This reproduces a source/discretization gap, not a failed Go calculation.
 if edgeThreshold(6,1,.9)!=3 || dm.txStates[a]!=types.StateSolid || dm.txStates[b]!=types.StateSolid || IsTournament(dm.graph){t.Fatal("expected fractional-boundary witness not reproduced")}
 t.Log("current matches literal Fig.1 but contradicts Lemma B.1: two solid vertices, no edge; n=6 F=1 gamma=.9, weights 2:2, threshold 3")
}

func TestIndependentReverseCanLeaveGraphUnchanged(t *testing.T) {
 a,b,c:=testTx(1),testTx(2),testTx(3)
 for attackers:=0;attackers<=4;attackers++ {
  reports:=make([]*types.LocalOrder,16)
  for i:=0;i<16;i++ {ids:=[][32]byte{a,b,c}; if i>=16-attackers{ids=[][32]byte{c,b,a}}; reports[i]=localOrder(uint64(i),ids...)}
  dm:=NewDependencyManager();dm.BuildGraphAndClassifyTxs(reports,20,4,1,nil);dm.CutShadedTail()
  if !graphHasEdge(dm.graph,a,b)||!graphHasEdge(dm.graph,a,c)||!graphHasEdge(dm.graph,b,c)||!reflect.DeepEqual(dm.ComputeFairOrder(),[][32]byte{a,b,c}){t.Fatalf("unexpected graph at adopted attackers=%d",attackers)}
  t.Logf("adopted attackers=%d: forward votes=%d, backward votes=%d<5, unchanged transitive graph and output A,B,C",attackers,16-attackers,attackers)
 }
}

 // Presence A=B=2, Z=4; weights AB=BA=1, AZ=ZA=BZ=ZB=2.
 // ID ties point A->Z and B->Z, so both shaded nodes reach solid Z.

func TestIndependentProposalBoundaryAndUpdateScope(t *testing.T){

}

func TestIndependentQueuedEvidenceBecomesInvalid(t *testing.T){
 auth:=newTestOrderAuthenticator(5);s,err:=NewOFOService(1,5,1,1,&testNetwork{},1,150,false,auth);if err!=nil{t.Fatal(err)};defer s.Stop()
 a,b,z,c:=testTx(1),testTx(2),testTx(3),testTx(4)

 if !s.validateReplicaOrders(queued){t.Fatal("fixture invalid before state change")}
 auditPartialFirst(t,s,auth)
 if s.validateReplicaOrders(queued){t.Fatal("expected stale-context rejection not reproduced")}
 rebased:=signedOrders(t,auth,5,1,2,map[uint64][][32]byte{0:{c}},map[uint64]map[uint64][][32]byte{1:{0:{a,b,z}}})[0]
 if !s.validateReplicaOrders(rebased){t.Fatal("new-context report should be valid")}
 t.Log("same valid signature and unused sequence: accepted before commit, rejected after commit; correctly regenerated List/Update passes. Do not bypass validation")
}

type auditCountingAuth struct{*testOrderAuthenticator;checks int}
func(a *auditCountingAuth)VerifyReplica(id uint64,digest [32]byte,sig []byte)bool{a.checks++;return a.testOrderAuthenticator.VerifyReplica(id,digest,sig)}

func TestIndependentRepeatedLeaderVerification(t *testing.T){
 base:=newTestOrderAuthenticator(5);auth:=&auditCountingAuth{testOrderAuthenticator:base};s,err:=NewOFOService(1,5,1,1,&testNetwork{},1,150,false,auth);if err!=nil{t.Fatal(err)};defer s.Stop()
 a:=testTx(1);reports:=signedOrders(t,base,5,1,1,map[uint64][][32]byte{0:{a},1:{a},2:{a},3:{a}},nil)
 for _,r:=range reports{if !s.validateReplicaOrders(r){t.Fatal("invalid fixture")}}
 p:=s.buildProposal(reports);if p==nil||!s.VerifyProposal(p){t.Fatal("build/verify failed")}
 if auth.checks!=12{t.Fatalf("signature checks=%d want12",auth.checks)}
 if !s.CommitVerifiedProposal(p)||auth.checks!=12{t.Fatal("verified commit repeated signature verification")}
 t.Log("collector-equivalent validation + build + hosting-equivalent verify: 12 checks for 4 signed lists; verified commit adds none")
}
